\section{Partitioned Regression}

\begin{frame}{Why Partition $\bm{X}$?}
\begin{itemize}
\item In applied work, we rarely care equally about every regressor. We typically have:
\begin{itemize}
\item A \textbf{treatment} or variable of interest ($\bm{X}_1$), and
\item \textbf{Controls} we include to avoid omitted variable bias ($\bm{X}_2$).
\end{itemize}\pause
\item Partitioning $\bm{X}=[\bm{X}_1\ \ \bm{X}_2]$ lets us answer three questions:
\begin{enumerate}
\item What is the formula for $\hat{\beta}_1$ \emph{holding $\bm{X}_2$ constant}? $\to$ Frisch-Waugh-Lovell.
\item What happens to $\hat{\beta}_1$ if we \emph{omit} $\bm{X}_2$? $\to$ Omitted variable bias formula.
\item How sensitive is $\hat{\beta}_1$ to \emph{unobserved} confounders? $\to$ Cinelli-Hazlett sensitivity analysis.
\end{enumerate}\pause
\item The algebra of partitioned matrices (Schur complement) is the common tool behind all three results.
\end{itemize}
\end{frame}


\begin{frame}{Recap: From Population to Sample Partitions}
In Lecture 2 we derived the partitioned regression result using population moments ($\mathbf{Q}$).
Now we apply the same logic to \textbf{sample} matrices:\pause
\begin{align*}
\bm{y}&=\bm{X}_1\bm{\beta}_1+\bm{X}_2\bm{\beta}_2+\bm{e},\qquad
\bm{X}=\begin{bmatrix}\bm{X}_1& \bm{X}_2\end{bmatrix}
\end{align*}\pause
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{lcc}
& \textbf{Population (Lecture 2)} & \textbf{Sample}\\
\hline
Moment matrix & $\mathbf{Q}_{XX}=\mathbb{E}[\bm{x}\bm{x}']$ & $\bm{X}'\bm{X}=\sum X_iX_i'$\\
Cross-moment & $\mathbf{Q}_{XY}=\mathbb{E}[\bm{x}Y]$ & $\bm{X}'\bm{y}=\sum X_iY_i$\\
Residual variation & $\mathbf{Q}_{11\cdot 2}$ & $\bm{X}_1'\bm{M}_2\bm{X}_1$\\
Key result & $\bm{\beta}_1=\mathbf{Q}_{11\cdot 2}^{-1}\mathbf{Q}_{1Y\cdot 2}$
  & $\hat{\bm{\beta}}_1=(\bm{X}_1'\bm{M}_2\bm{X}_1)^{-1}\bm{X}_1'\bm{M}_2\bm{y}$
\end{tabular}
\end{center}
\onslide<4->{%
where $\bm{M}_2=\bm{I}-\bm{P}_2$ and $\bm{P}_2=\bm{X}_2(\bm{X}_2'\bm{X}_2)^{-1}\bm{X}_2'$.}
\end{frame}


\begin{frame}{Why Open Up $(\bm{X}'\bm{X})^{-1}$?}
\begin{itemize}
\item Every OLS coefficient can be traced back to specific variation in the data.\pause
\item $\bm{\hat{\beta}}_1$ does \emph{not} use all the variation in $\bm{X}_1$, only the part that the other regressors cannot explain.\pause
\item \alert{The Schur complement tells you exactly which variation in the data identifies each coefficient.}
\end{itemize}
\pause
\begin{tcolorbox}[colback=blue!5, colframe=blue!50, boxsep=2pt, top=2pt, bottom=2pt]
\textbf{Definition.} For a conformably partitioned matrix
$\bm{A}=\begin{bmatrix}\bm{A}_{11}&\bm{A}_{12}\\\bm{A}_{21}&\bm{A}_{22}\end{bmatrix}$
with $\bm{A}_{22}$ invertible, the \textbf{Schur complement} of $\bm{A}_{22}$ in $\bm{A}$ is
$$\bm{A}/\bm{A}_{22}\;\equiv\;\bm{A}_{11}-\bm{A}_{12}\bm{A}_{22}^{-1}\bm{A}_{21}.$$
\end{tcolorbox}
\end{frame}


\begin{frame}{Schur Complement: A $2\times 2$ Example}
\begin{itemize}\setlength{\itemsep}{2pt}
\item Two scalar regressors (education, experience):
$\bm{X}'\bm{X}=\begin{psmallmatrix} 10 & 3 \\ 3 & 8 \end{psmallmatrix}$\pause
\item Direct inversion: $(\bm{X}'\bm{X})^{-1}=\frac{1}{10\cdot 8-3^2}\begin{psmallmatrix} 8 & -3 \\ -3 & 10 \end{psmallmatrix}=\frac{1}{71}\begin{psmallmatrix} 8 & -3 \\ -3 & 10 \end{psmallmatrix}$\pause
\item Schur complement for $B_{11}$: $\left(10-\frac{3^2}{8}\right)^{-1}=\left(\frac{71}{8}\right)^{-1}=\frac{8}{71}$ \checkmark\pause
\item The denominator $10-\frac{9}{8}=\frac{71}{8}$ is the variance of education \emph{not explained by} experience.
\end{itemize}
\begin{tcolorbox}[colback=blue!5, colframe=blue!50, boxsep=2pt, top=2pt, bottom=2pt]
The Schur complement isolates the \emph{unique} information in $\bm{X}_1$ after removing $\bm{X}_2$. More collinearity $\to$ less unique info $\to$ larger standard errors.
\end{tcolorbox}
\end{frame}

\begin{frame}{Intuition: What Does the Schur Complement Remove?}
\vspace{-0.5em}
\includegraphics[width=1.8in]{Bal1.pdf}
\vspace{-0.3em}
\begin{align*}
 (\bm{X}_1'\bm{X}_1- \bm{X}_1'\bm{P}_2\bm{X}_1)^{-1}&=  (\bm{X}_1'\bm{X}_1- \bm{X}_1'\bm{P}_2\bm{P}_2\bm{X}_1)^{-1}\\
&=(\underbrace{\bm{X}_1'\bm{X}_1}_{\text{variance of }\bm{X}_1}- \underbrace{\bm{X}_1'\bm{P}_2\bm{P}_2\bm{X}_1}_{\text{Projection of }\bm{X}_1\text{ onto the column space of }\bm{X}_2})^{-1}
\end{align*}
\end{frame}


\begin{frame}{The Sample Block Inverse}
Applying the block inverse (derived in Lecture 2) to $\bm{X}'\bm{X}$:\pause
$$(\bm{X}'\bm{X})^{-1}=\begin{bmatrix}
  (\bm{X}_1'\bm{M}_2\bm{X}_1)^{-1}
  & -(\bm{X}_1'\bm{M}_2\bm{X}_1)^{-1}\bm{X}_1'\bm{X}_2(\bm{X}_2'\bm{X}_2)^{-1}\\[6pt]
  -(\bm{X}_2'\bm{M}_1\bm{X}_2)^{-1}\bm{X}_2'\bm{X}_1(\bm{X}_1'\bm{X}_1)^{-1}
  & (\bm{X}_2'\bm{M}_1\bm{X}_2)^{-1}
\end{bmatrix}$$\pause
\begin{itemize}
\item The diagonal blocks are \textbf{Schur complements}: residual variation after projecting out the other group.\pause
\item Multiplying by $\bm{X}'\bm{y}$ recovers:
$$\hat{\bm{\beta}}_1=(\bm{X}_1'\bm{M}_2\bm{X}_1)^{-1}\bm{X}_1'\bm{M}_2\bm{y}$$
\end{itemize}
\end{frame}



\section{Proof of FWL}
\setcounter{subsection}{1}
\begin{frame}{Frisch-Waugh-Lovell: The Claim}
\begin{itemize}
\item FWL claim: the coefficient $\bm{\hat{\beta}}_2$ from the full regression is \emph{identical} to the coefficient from regressing residualized $\bm{y}$ on residualized $\bm{X}_2$.\pause
\item That is, we can first project $\bm{y}$ and $\bm{X}_2$ into the orthogonal complement of the column space of $\bm{X}_1$, then regress one on the other.\pause
\item Recall: $\bm{M}_1 = \bm{I} - \bm{X}_1(\bm{X}_1'\bm{X}_1)^{-1}\bm{X}_1'$ is the ``residual maker'': $\bm{M}_1\bm{q}$ produces the residuals of any variable $\bm{q}$ regressed on $\bm{X}_1$.
\end{itemize}
\end{frame}

\begin{frame}{Frisch-Waugh-Lovell: Removing $\bm{X}_1$}
Start from the full model and premultiply both sides by $\bm{M}_1$:
\begin{align*}
\bm{y}&=\bm{X}_1 \bm{\hat{\beta}}_1+\bm{X}_2 \bm{\hat{\beta}}_2+\bm{e} \tag{full model}\\\pause
\bm{M}_1\bm{y}&=\bm{M}_1\bm{X}_1 \bm{\hat{\beta}}_1+\bm{M}_1 \bm{X}_2 \bm{\hat{\beta}}_2+\bm{M}_1\bm{e} \tag{premultiply by $\bm{M}_1$}\\\pause
\bm{M}_1\bm{y}&=\bm{M}_1 \bm{X}_2 \bm{\hat{\beta}}_2+\bm{M}_1\bm{e} \tag{$\bm{M}_1\bm{X}_1=\bm{0}$ by construction}
\end{align*}\pause
\begin{tcolorbox}[colback=blue!5, colframe=blue!50, boxsep=2pt, top=2pt, bottom=2pt]
\textbf{One trick:} $\bm{M}_1$ annihilates $\bm{X}_1$, so $\hat{\beta}_1$ drops out entirely.
\end{tcolorbox}
\end{frame}

\begin{frame}{Frisch-Waugh-Lovell: Solving for $\hat{\beta}_2$}
The residualized equation $\bm{M}_1\bm{y}=\bm{M}_1 \bm{X}_2 \bm{\hat{\beta}}_2+\bm{M}_1\bm{e}$ is just a regression---apply OLS:
\begin{align*}
\bm{\hat{\beta}}_2&=((\bm{M}_1\bm{X}_2)'\bm{M}_1\bm{X}_2)^{-1}(\bm{M}_1\bm{X}_2)'\bm{M}_1\bm{y} \tag{OLS on residualized data}\\\pause
&=(\bm{X}_2'\bm{M}_1'\bm{M}_1\bm{X}_2)^{-1}\bm{X}'_2\bm{M}_1'\bm{M}_1\bm{y} \tag{expand transpose}\\\pause
&=(\bm{X}_2'\bm{M}_1\bm{X}_2)^{-1}\bm{X}'_2\bm{M}_1\bm{y} \tag{$\bm{M}_1'=\bm{M}_1$, \ $\bm{M}_1\bm{M}_1=\bm{M}_1$}
\end{align*}\pause
\begin{tcolorbox}[colback=blue!5, colframe=blue!50, boxsep=2pt, top=2pt, bottom=2pt]
\textbf{One trick:} $\bm{M}_1$ is symmetric and idempotent, so $\bm{M}_1'\bm{M}_1 = \bm{M}_1$.
\end{tcolorbox}
\end{frame}

\begin{frame}{FWL Proof Part I: Deriving $\hat{\beta}_1$}
Start from the normal equations $\bm{X'X}\hat{\beta}=\bm{X'y}$ and take the first row:
\begin{align*}
(\bm{X}_1'\bm{X}_1)\bm{\hat{\beta}}_1+(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2&=\bm{X}'_1\bm{y} \tag{1st normal equation}\\[4pt]
(\bm{X}_1'\bm{X}_1)\bm{\hat{\beta}}_1&=\bm{X}'_1\bm{y}-(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2 \tag{isolate $\hat{\beta}_1$}\\[4pt]
\bm{\hat{\beta}}_1&=\underbrace{(\bm{X}_1'\bm{X}_1)^{-1}\bm{X}'_1\bm{y}}_{\hat{\beta}_{res}}-\underbrace{(\bm{X}_1'\bm{X}_1)^{-1}(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2}_{\text{OVB correction}} \tag{premultiply by $(\bm{X}_1'\bm{X}_1)^{-1}$}\\[4pt]
\bm{\hat{\beta}}_1&=(\bm{X}_1'\bm{X}_1)^{-1}\bm{X}'_1(\bm{y}-\bm{X}_2\bm{\hat{\beta}}_2) \tag{factor: $\hat{\beta}_1$ from residualized $\bm{y}$}
\end{align*}
\end{frame}

\begin{frame}{FWL Proof Part II: Deriving $\hat{\beta}_2$ (Setup)}
Start from the second row of the normal equations and substitute $\hat{\beta}_1$:
\begin{align*}
(\bm{X}_2'\bm{X}_1)\bm{\hat{\beta}}_1+(\bm{X}_2'\bm{X}_2)\bm{\hat{\beta}}_2&=\bm{X}'_2\bm{y} \tag{2nd normal equation}\\[4pt]
(\bm{X}_2'\bm{X}_1)[(\bm{X}_1'\bm{X}_1)^{-1}\bm{X}'_1(\bm{y}-\bm{X}_2\bm{\hat{\beta}}_2)]+(\bm{X}_2'\bm{X}_2)\bm{\hat{\beta}}_2&=\bm{X}'_2\bm{y} \tag{plug in $\hat{\beta}_1$ from Part I}\\[4pt]
\bm{X}_2'\bm{P}_1\bm{y}-\bm{X}'_2\bm{P}_1\bm{X}_2\bm{\hat{\beta}}_2+(\bm{X}_2'\bm{X}_2)\bm{\hat{\beta}}_2&=\bm{X}'_2\bm{y} \tag{recognize $\bm{P}_1$}
\end{align*}
\end{frame}

\begin{frame}{FWL Proof Part II: Deriving $\hat{\beta}_2$ (Result)}
Continuing---collect $\hat{\beta}_2$ terms on the left, everything else on the right:
\begin{align*}
(\bm{X}_2'\bm{X}_2-\bm{X}'_2\bm{P}_1\bm{X}_2)\bm{\hat{\beta}}_2&=\bm{X}'_2\bm{y}-\bm{X}_2'\bm{P}_1\bm{y} \tag{rearrange}\\[4pt]
\bm{X}_2'(\bm{I}-\bm{P}_1)\bm{X}_2\bm{\hat{\beta}}_2&=\bm{X}'_2(\bm{I}-\bm{P}_1)\bm{y} \tag{factor out $(\bm{I}-\bm{P}_1)$}\\[4pt]
\bm{X}_2'\bm{M}_1\bm{X}_2\bm{\hat{\beta}}_2&=\bm{X}'_2\bm{M}_1\bm{y} \tag{$\bm{I}-\bm{P}_1=\bm{M}_1$}\\[4pt]
\bm{\hat{\beta}}_2&=(\bm{X}_2'\bm{M}_1\bm{X}_2)^{-1}\bm{X}'_2\bm{M}_1\bm{y} \tag{solve for $\hat{\beta}_2$}
\end{align*}
This is the FWL result: $\hat{\beta}_2$ from the full regression equals the coefficient from regressing $\bm{M}_1\bm{y}$ on $\bm{M}_1\bm{X}_2$ (residual on residual).
\end{frame}


\begin{frame}{Frisch-Waugh-Lovell (FWL)}
\begin{itemize}
\item Under $\bm{y} = \bm{X\beta}+\bm{e}$,
\begin{align*}
\bm{\hat{y}} &=b_0+ b_1\bm{x}_1+\ldots + b_{j-1} \bm{x}_{j-1} + b_{j} \bm{x}_{j}\quad + b_{j+1} \bm{x}_{j+1}+  \ldots+b_{K} \bm{x}_{K}\\\pause
\bm{\tilde{y}} &= d_0+ d_1\bm{x}_1+\ldots + d_{j-1} \bm{x}_{j-1} + \quad0\quad \ + d_{j+1} \bm{x}_{j+1}+  \ldots+d_{K} \bm{x}_{K}\\
\bm{\hat{x}_j} &= c_0+ c_1\bm{x}_1+\ldots + c_{j-1} \bm{x}_{j-1}+ \quad0\quad \ +  c_{j+1} \bm{x}_{j+1}+ \ldots+c_{K} \bm{x}_{K}\pause
\end{align*}
\item FWL: The regression coefficient $b_j$ is equivalent to a regression coefficient $b^{*}_1$ produced by regressing the residualised outcome $\bm{e_y}=\bm{y}-\bm{\tilde{y}}$ on the residualised $\bm{x_j}$: $\bm{e_{x_j}}=\bm{x_j}-\bm{\hat{x}}_j$.
$$\bm{e}_y= b^*_0+ b^*_1\bm{e}_{x_j}+\bm{e}$$
\end{itemize}
\end{frame}

\section{Leverage}

\begin{frame}{Trace of Projection and Annihilator Matrix}
\begin{itemize}
\item Recall, the \textbf{trace} of a matrix is the sum of the diagonal elements and the sum of the eigenvalues.
\item The \textbf{rank} of a matrix is the maximum number of linearly independent column vectors.
\item The trace of $\bm{P}$ is its rank $k$
\begin{itemize}
\item The projection matrix is \textbf{idempotent}: $\bm{P}=\bm{P}^2$.
\item If $\lambda$ is an eigenvalue of $\bm{P}$, $\bm{P}v=\lambda v$
\item Applying $\bm{P}$ again: $\bm{P}^2v=\bm{P}(\bm{P}v)=\bm{P}(\lambda v)=\lambda^2 v$.
\item But $\bm{P}^2=\bm{P}$, so $\lambda^2 v=\bm{P}v=\lambda v$, giving $\lambda^2=\lambda$.  Only true for $\lambda=0$ or $\lambda=1$.
\item The eigenvalue 1 corresponds to a direction preserved by the projection, the dimension of the column space of $\bm{X}$.
\end{itemize}
\item $tr(\bm{M})=tr(\bm{I})-tr(\bm{P})=n-k$
\end{itemize}
\end{frame}

\begin{frame}{Alternative proof that Trace of Projection is $k$}
\begin{itemize}
\item Theorem: $\text{tr} \bm{P}=k$.
\begin{align*}
\text{tr}\bm{P}&=\text{tr}\left(\bm{X}(\bm{X}'\bm{X})^{-1}\bm{X}'\right)\\
&=\text{tr}\left((\bm{X}'\bm{X})^{-1}\bm{X}'\bm{X}\right)\\
&=\text{tr}(\bm{I}_k)\\
&=k
\end{align*}
\end{itemize}
\end{frame}

\begin{frame}{Leverage}
\begin{itemize}
\item The $i$'th diagonal element of $\bm{P}$ is $h_{ii}=X_i'(\bm{X}'\bm{X})^{-1}X_i$ is called the \emph{leverage} of the $i$'th observation. 
\item $h_{ii}$ ranges between 0 and 1, measures how unusual the $i^{th}$ observation $X_i$ is relative to other observations.
\item A regression design is called \textbf{balanced} when the leverage values are roughly equal.  
\item Recall, the sum of $h_{ii}=k$
\item A regression is perfectly balanced if $\max(h_{ii})=k/n$
\end{itemize}
\end{frame}


\begin{frame}{Leave-One-Out Regression}
\small
\begin{itemize}\itemsep=2pt
\item The Leave-One-Out Regression estimates the model excluding observation $i$:
\vspace{-0.3em}
\begin{align*}
\hat{\beta}_{(-i)}&=\left( \sum_{j\neq i}X_jX_j'\right)^{-1}\left( \sum_{j\neq i}X_jY_j\right)
=\left(\bm{X'X}-X_iX_i'\right)^{-1}\left( \bm{X'y}-X_iY_i\right)
\end{align*}
\vspace{-0.8em}
\item $\hat{\beta}_{(-i)}$ is not a function of $(Y_i, X_i)$, so it can be used for prediction: $\tilde{Y}_i=X_i'\hat{\beta}_{(-i)}$.
\item The \textbf{leave-one-out residual} (prediction error) is $\tilde{e}_i=Y_i-\tilde{Y}_i$.
\end{itemize}
\end{frame}

\begin{frame}{Leave-One-Out Regression}
\begin{itemize}
\item Calculating $\hat{\beta}_{(-i)}$ comes for free from our projection model:
$$\hat{\beta}_{(-i)}=\hat{\beta}-(\bm{X}'\bm{X})^{-1}X_i\tilde{e}_i$$
$$\tilde{e}_i=(1-h_{ii})^{-1}\hat{e}_i$$
\item We can define a modified annihilator matrix: 
$$\bm{M}^*\equiv (\bm{I}_n-\text{diag}\{h_{11},\ldots,h_{nn}\})^{-1}$$
Which allows us to rewrite:
$$\tilde{\bm{e}}=\bm{M}^*\hat{\bm{e}}$$
\item We observe the residuals $\hat{\bm{e}}$ and we know how much each observation affects the regression, so we can just subtract that out.
\end{itemize}
\end{frame}


\begin{frame}{Deriving the Leave-One-Out Formula (Step 1)}
Goal: express $\hat{\beta}_{(-i)}$ in terms of $\hat{\beta}$ and $\tilde{e}_i$. Start from the definition:
\begin{align*}
\hat{\beta}_{(-i)}&=(\bm{X}'\bm{X}-X_iX_i')^{-1}(\bm{X}'\bm{y}-X_iY_i) \tag{definition}\\[4pt]
(\bm{X}'\bm{X}-X_iX_i')\hat{\beta}_{(-i)}&=\bm{X}'\bm{y}-X_iY_i \tag{multiply both sides by $(\cdots)$}\\[4pt]
\bm{X}'\bm{X}\hat{\beta}_{(-i)}-X_iX_i'\hat{\beta}_{(-i)}&=\bm{X}'\bm{y}-X_iY_i \tag{distribute}\\[4pt]
\hat{\beta}_{(-i)}-(\bm{X}'\bm{X})^{-1}X_iX_i'\hat{\beta}_{(-i)}&=\hat{\beta}-(\bm{X}'\bm{X})^{-1}X_iY_i \tag{premultiply by $(\bm{X'X})^{-1}$}
\end{align*}
\end{frame}

\begin{frame}{Deriving the Leave-One-Out Formula (Step 2)}
Continuing from the previous slide:
\begin{align*}
\hat{\beta}_{(-i)}&=\hat{\beta}-(\bm{X}'\bm{X})^{-1}X_iY_i+(\bm{X}'\bm{X})^{-1}X_iX_i'\hat{\beta}_{(-i)} \tag{move $\hat{\beta}_{(-i)}$ terms right}\\[4pt]
&=\hat{\beta}-(\bm{X}'\bm{X})^{-1}X_i\bigl(Y_i-X_i'\hat{\beta}_{(-i)}\bigr) \tag{factor out $(\bm{X'X})^{-1}X_i$}\\[4pt]
&=\hat{\beta}-(\bm{X}'\bm{X})^{-1}X_i\,\tilde{e}_i \tag{recognize $\tilde{e}_i=Y_i-X_i'\hat{\beta}_{(-i)}$}
\end{align*}
\vspace{0.5em}
The leave-one-out estimate equals the full-sample estimate minus a correction proportional to the prediction error $\tilde{e}_i$.
\end{frame}

\begin{frame}{Connecting $\hat{e}_i$ to the Prediction Error $\tilde{e}_i$ (Setup)}
Start from the LOO result. Premultiply by $X_i'$ to get predictions:
\begin{align*}
\hat{\beta}_{(-i)}&=\hat{\beta}-(\bm{X}'\bm{X})^{-1}X_i\tilde{e}_i \tag{from previous slide}\\[4pt]
X_i'\hat{\beta}_{(-i)}&=X_i'\hat{\beta}-X_i'(\bm{X}'\bm{X})^{-1}X_i\tilde{e}_i \tag{premultiply by $X_i'$}\\[4pt]
X_i'\hat{\beta}_{(-i)}&=X_i'\hat{\beta}-h_{ii}\tilde{e}_i \tag{recognize $h_{ii}=X_i'(\bm{X'X})^{-1}X_i$}
\end{align*}
\end{frame}

\begin{frame}{Connecting $\hat{e}_i$ to the Prediction Error $\tilde{e}_i$ (Result)}
Now subtract both sides from $Y_i$:
\begin{align*}
Y_i-X_i'\hat{\beta}_{(-i)}&=Y_i-X_i'\hat{\beta}+h_{ii}\tilde{e}_i \tag{subtract from $Y_i$}\\[4pt]
\tilde{e}_i&=\hat{e}_i+h_{ii}\tilde{e}_i \tag{recognize $\tilde{e}_i$ and $\hat{e}_i$}\\[4pt]
\tilde{e}_i(1-h_{ii})&=\hat{e}_i \tag{collect $\tilde{e}_i$ terms}\\[4pt]
\tilde{e}_i&=(1-h_{ii})^{-1}\hat{e}_i \tag{solve for $\tilde{e}_i$}
\end{align*}
High-leverage observations ($h_{ii}$ close to 1) inflate the prediction error relative to the residual.
\end{frame}


\begin{frame}{Check your understanding}
\begin{itemize}
\item For what observation would $\hat{\beta}_{(-i)}=\hat{\beta}$?
\end{itemize}
\end{frame}

\begin{frame}{Influential observations}
\begin{itemize}
\item An observation $i$ is influential if its omission changes the parameter of interest.
\item $\hat{\beta}- \hat{\beta}_{(-i)}=(\bm{X}'\bm{X})^{-1}X_i\tilde{e}_i$.
\item Premultiply by $X_i'$, and we get that 
$$\hat{Y}_i-\tilde{Y}_i=X'_i(\bm{X}'\bm{X})^{-1}X_i\tilde{e}_i=h_{ii}\tilde{e}_i$$
\item So $i$ is influential if $h_{ii}$ is big and $|\tilde{e}_i|$ is big.
\item This warrants investigation: it could indicate a data entry error, an outlier from a different population, or a genuine extreme case that deserves scrutiny.
\end{itemize}
\end{frame}

\section{Bias}


\begin{frame}{What Happens When We Leave Something Out?}
\begin{itemize}
\item So far: we derived $\hat{\beta}$ assuming we included the right regressors.\pause
\item In practice, we always leave something out. The question is whether it matters.\pause
\item An estimator is \textbf{biased} if $E[\hat{\beta}|\bm{X}]\neq\beta$: on average across samples, we get the wrong answer.\pause
\item The FWL result gives us the tool to quantify this: we already showed that $\hat{\beta}_1$ depends on what else is in the model. Now we ask: what if $\bm{X}_2$ \emph{should} be in the model but isn't?
\end{itemize}
\end{frame}


\begin{frame}{Example: Factory Closures and Voting Behavior}
  \begin{itemize}
    \item A factory closure is visible evidence that trade may harm the local economy. Does it push voters toward protectionism?\pause
    \item Suppose we regress voting on factory closures:
  \[
  Y_i = \alpha + \beta\,\text{Closure}_i + \epsilon_i
  \]
    \item But closures don't happen at random---they concentrate in regions with specific \emph{industrial compositions}. And industrial composition independently affects voting.\pause
    \item The full model should include both:
  \[
  Y_i = \alpha + \beta\,\text{Closure}_i + \gamma\,\text{IndComp}_i + \delta\,(\text{Closure}_i \times \text{IndComp}_i) + \epsilon_i
  \]
    \item If we omit IndComp, our estimate of $\beta$ absorbs part of $\gamma$. How much?
  \end{itemize}
\end{frame}


\begin{frame}{The Omitted Variable Bias Formula}
Recall the normal equations with $\bm{X}=[\bm{X}_1\ \ \bm{X}_2]$:
$$\begin{bmatrix} \bm{X}_1'\bm{X}_1& \bm{X}_1'\bm{X}_2\\  \bm{X}_2'\bm{X}_1 &\bm{X}_2'\bm{X}_2 \end{bmatrix} \begin{bmatrix}\bm{\hat{\beta}}_1\\ \bm{\hat{\beta}}_2\end{bmatrix}=\begin{bmatrix}\bm{X}_1'\bm{y}\\ \bm{X}_2'\bm{y}\end{bmatrix}$$\pause
Derivation of $\bm{\hat{\beta}}_1$:
\begin{align*}
(\bm{X}_1'\bm{X}_1)\bm{\hat{\beta}}_1+(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2&=\bm{X}'_1\bm{y}\\\pause
(\bm{X}_1'\bm{X}_1)\bm{\hat{\beta}}_1&=\bm{X}'_1\bm{y}-(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2\\\pause
\bm{\hat{\beta}}_1&=(\bm{X}_1'\bm{X}_1)^{-1}\bm{X}'_1\bm{y}-(\bm{X}_1'\bm{X}_1)^{-1}(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2\\ \pause
\bm{\hat{\beta}}_{res}&=\bm{\hat{\beta}}_1+\underbrace{(\bm{X}_1'\bm{X}_1)^{-1}(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2}_{\widehat{bias}\ for\ excluding\ X_2}
\end{align*}
\end{frame}



\begin{frame}{Unpacking the Bias: Imbalance $\times$ Impact}

$$\widehat{\text{bias}} = \underbrace{(\bm{X}_1'\bm{X}_1)^{-1}(\bm{X}_1'\bm{X}_2)}_{\text{Imbalance: how well does }X_1\text{ predict }X_2}\times\underbrace{\bm{\hat{\beta}}_2}_{\text{Impact: effect of }X_2\text{ on }Y}$$
\begin{itemize}
\item $(\bm{X}_1'\bm{X}_1)^{-1}(\bm{X}_1'\bm{X}_2)$ are the coefficients from regressing $\bm{X}_2$ on $\bm{X}_1$: how much do the confounders differ across treatment groups?\pause
\item $\bm{\hat{\beta}}_2$ is how much $X_2$ matters for $Y$, holding $X_1$ constant.\pause
\item Bias is large only when \emph{both} are large: a confounder must be imbalanced \emph{and} consequential.\pause
\item The problem: $\bm{X}_1$ and $\bm{X}_2$ are multivariate, so the bias is a matrix product that is hard to sign. We need a scalar summary.
\end{itemize}
\end{frame}

\section{Partial \texorpdfstring{$R^2$}{R-squared}}

\begin{frame}{From Matrix Bias to a Scalar: Partial $R^2$}
\begin{itemize}
\item Cinelli and Hazlett (2020) propose studying omitted variables using $R^2$ as a metric.
\item Suppose the full linear regression is as follows:
$$Y=\bm{X}_1\bm{\beta}_1+\bm{X}_2\bm{\beta}_2+\bm{e}$$
\item $\bm{P}_1=\bm{X}_1(\bm{X}'_1\bm{X}_1)^{-1}\bm{X}_1'$ is the projection matrix onto $\bm{X}_1$.
\item $\bm{M}_1=\bm{I}-\bm{X}_1(\bm{X}'_1\bm{X}_1)^{-1}\bm{X}_1'$ is the residual maker for $\bm{X}_1$.
\item In general the partial $R^2$ measures the proportion of the variance in $Y$ that is uniquely explained by a set of predictors $\bm{X}_2$ after accounting for $\bm{X}_1$.
\end{itemize}
\end{frame}


\begin{frame}{Formula for $R^2$ with one variable}
\begin{align*}
R^2_{Y\sim \bm{Z}} &= \frac{SSR}{TSS} &&\text{(definition (1) of $R^2$)}\\[10pt]
&= \frac{var(\hat{Y})}{var(Y)} &&\text{($SSR$ is variance explained)}\\[10pt]
&= 1 - \frac{var(e)}{var(Y)} &&\text{(since $var(Y)=var(\hat{Y})+var(e)$)}\\[10pt]
&= corr(Y, \hat{Y})^2 &&\text{(definition (2) of $R^2$)}\\[10pt]
&= corr(Y, \bm{Z})^2 &&\text{($\hat{Y}$ is linear prediction from $\bm{Z}$)}
\end{align*}

\end{frame}


\begin{frame}[fragile]{Example in R}
\begin{lstlisting}
library(carData)
attach(Prestige)
mod <- lm(prestige ~ education)
summary(mod)$r.square
# [1] 0.7228007
var(predict(mod)) / var(prestige) 
1 - var(resid(mod)) / var(prestige) 
cor(prestige, predict(mod))^2
cor(prestige, education)^2
\end{lstlisting}

\end{frame}


\begin{frame}{Formula for partial $R^2$}
\begin{itemize}
\item Using Residual Sum of Squares:
\begin{itemize}
\item $R^2_{\bm{Y}\sim \bm{X}_2|\bm{X}_1}=1-\frac{\text{RSS}(\bm{X}_1,\bm{X}_2)}{\text{RSS}(\bm{X}_1)}$
\item $\text{RSS}(\bm{X}_1)=||\bm{M}_1Y||^2=Y'\bm{M_1}'\bm{M}_1Y=Y'\bm{M_1}Y$
\item $\text{RSS}(\bm{X}_{1},\bm{X}_2)=||\bm{M}_{1,2}Y||^2=Y'\bm{M}_{1,2}Y$
\end{itemize}\pause
\item Using Sum of Squared Residuals:
\begin{itemize}
\item $R^2_{\bm{Y}\sim \bm{X}_2|\bm{X}_1}=\frac{\text{SSR}(\bm{X}_2|\bm{X}_1)}{\text{RSS}(\bm{X}_1)}$
\item SSR, the explained sum of squares, the amount explained by $\bm{X}_2$ once accounting for $\bm{X}_1$
\begin{align*}
\text{SSR}(\bm{X}_2|\bm{X}_1)&=Y'\bm{M}_1\bm{X}_2(\bm{X}_2'\bm{M}_1\bm{X}_2)^{-1}\bm{X}_2'\bm{M}_1Y\\
&=Y'\bm{M}_1\bm{X}_2\bm{\hat{\beta}}_2
\end{align*}
\end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{Cinelli Hazlett (2020) formulation}
We model $Y=\hat{\beta}_1D+e$, so $\hat{\beta}_1=\frac{cov(D,\ Y)}{var(D)}$.\\
\bigskip
What if there is regression of $Y$ on $D$ and there is a single omitted variable $Z$?
\begin{align*}
\widehat{\text{bias}} &= (\bm{X}_1'\bm{X}_1)^{-1}(\bm{X}_1'\bm{X}_2)\bm{\hat{\beta}}_2\\
 &=\left(\frac{cov(D, Z)}{var(D)}\right)\left(\frac{cov(Z^{\perp D},\ Y^{\perp D})}{var(Z^{\perp D})}\right)
 \end{align*}
 Lets write this in terms of $D$ and two new formalisms for the disturbance: $Z^{\perp D}$ the part of $Z$ not predicted by $D$, and $Y^{\perp D}$ the part of $Y$ not predicted by $D$. 
\end{frame}

\begin{frame}{Bias in Terms of Covariances}
We start with the regression model:
\[
Y = \hat{\beta}_1 D + e,\qquad \hat{\beta}_1 = \frac{\operatorname{cov}(D,Y)}{\operatorname{var}(D)}.
\]
\vspace{1ex}
Now consider the case where a variable \(Z\) is omitted from the regression of \(Y\) on \(D\).  
The omitted variable bias is given by:
\[
\widehat{\text{bias}} = \frac{\operatorname{cov}(D,Z)}{\operatorname{var}(D)}\cdot\frac{\operatorname{cov}(Z^{\perp D},Y^{\perp D})}{\operatorname{var}(Z^{\perp D})}.
\]
\textbf{Note:}  
The first fraction reflects how \(D\) and \(Z\) co-move, while the second captures the effect of \(Z\) on \(Y\) once \(D\)’s influence is removed.
\end{frame}

\begin{frame}{Converting Covariances to Correlations}
\vspace{-1ex}
\textbf{For the First Term:}
\[
\frac{\operatorname{cov}(D,Z)}{\operatorname{var}(D)} 
=\frac{\operatorname{corr}(D,Z)\,sd(D)\,sd(Z)}{sd(D)^2}
=\frac{\operatorname{corr}(D,Z)\,sd(Z)}{sd(D)}.
\]
\textbf{Note:}  
We used the identity
\[
\operatorname{cov}(X,Y)=\operatorname{corr}(X,Y)\,sd(X)\,sd(Y),
\]
and noted that \(\operatorname{var}(D)=sd(D)^2\).

\vspace{2ex}\pause
\textbf{For the Second Term:}
\[
\frac{\operatorname{cov}(Z^{\perp D},Y^{\perp D})}{\operatorname{var}(Z^{\perp D})}
=\frac{\operatorname{corr}(Z^{\perp D},Y^{\perp D})\,sd(Z^{\perp D})\,sd(Y^{\perp D})}{sd(Z^{\perp D})^2}
=\frac{\operatorname{corr}(Z^{\perp D},Y^{\perp D})\,sd(Y^{\perp D})}{sd(Z^{\perp D})}.
\]
\textbf{Note:}  
Similarly, we express the covariance in terms of correlation and standard deviations, with \(\operatorname{var}(Z^{\perp D})=sd(Z^{\perp D})^2\).
\end{frame}

\begin{frame}{The Bias Formula in Terms of Partial $R^2$}
Combining the two unpacked terms, we obtain:
\[
\widehat{\text{bias}} 
=\left(\frac{\operatorname{corr}(D,Z)\,sd(Z)}{sd(D)}\right)
\left(\frac{\operatorname{corr}(Z^{\perp D},Y^{\perp D})\,sd(Y^{\perp D})}{sd(Z^{\perp D})}\right).
\]
Rearranging, this becomes:
\[
\widehat{\text{bias}} = \frac{\operatorname{corr}(D,Z)\,\operatorname{corr}(Z^{\perp D},Y^{\perp D})\,sd(Z)\,sd(Y^{\perp D})}{sd(D)\,sd(Z^{\perp D})}.
\]
Replacing partial $R^2$:
\[
|\widehat{\text{bias}}| = \sqrt{\frac{R^2_{D\sim Z}\,R^2_{Y\sim Z|D}}{1 - R^2_{D\sim Z}}}\cdot\frac{sd(Y^{\perp D})}{sd(D)}.
\]
\end{frame}

\begin{frame}{What Does the Bias Formula Tell Applied Researchers?}
\begin{itemize}
\item The bias depends on two quantities, both expressed as partial $R^2$:
\begin{enumerate}
\item $R^2_{D\sim Z}$: How strongly does the omitted variable predict the \textbf{treatment}?
\item $R^2_{Y\sim Z|D}$: How strongly does the omitted variable predict the \textbf{outcome}, after accounting for treatment?
\end{enumerate}\pause
\item If \emph{either} partial $R^2$ is small, the bias is small---a confounder must predict both $D$ and $Y$ to be dangerous.\pause
\item In practice, you ask: ``Is it plausible that an unobserved variable explains $x\%$ of the residual variation in $D$ and $y\%$ of the residual variation in $Y$?''
\item Observed covariates serve as benchmarks: if the strongest observed predictor has $R^2_{Y\sim Z|D}=0.05$, an omitted confounder would need to be far stronger to overturn your result.
\end{itemize}
\end{frame}

% Slide 1: Simulation Setup and Motivation
\begin{frame}{Example: Simulation Setup for Sensitivity Analysis}
\textbf{Simulation Design:}
\begin{itemize}
    \item Generate \(N\) observations with treatment \(D\), observed confounder \(Z\), and outcome:
    \[
    Y = \beta_D D + \beta_Z Z + e,\quad e \sim N(0,1).
    \]
    \item We first show that omitting \(Z\) biases the estimate of \(\beta_D\).
    \item Then we include \(Z\) and ask: how robust is our estimate to \emph{additional} unobserved confounders?
\end{itemize}
\end{frame}

% Slide 2: Data Simulation and Comparing Models
\begin{frame}[fragile]{Step 1: Simulate Data and Compare Models}
\begin{lstlisting}
set.seed(123)
N <- 1000; beta_D <- 2; beta_Z <- 3
D <- rbinom(N, 1, 0.5); Z <- rnorm(N)
Y <- beta_D * D + beta_Z * Z + rnorm(N)

# Naive model (omitting Z) -- biased
coef(lm(Y ~ D))          # beta_D != 2

# Full model (including Z) -- unbiased
model_full <- lm(Y ~ D + Z)
coef(model_full)          # beta_D ~ 2
\end{lstlisting}
\vspace{1ex}
The naive model is biased because $Z$ is correlated with both $D$ and $Y$. The full model recovers the true effect---but what if there were \emph{another} confounder we could not observe?
\end{frame}

% Slide 3: Conducting Sensitivity Analysis with sensemakr
\begin{frame}[fragile]{Step 2: Running the Sensitivity Analysis}
\begin{lstlisting}
library(sensemakr)

# Sensitivity analysis on the FULL model
sensitivity <- sensemakr(model_full, treatment = "D",
                         benchmark_covariates = "Z")
summary(sensitivity)
plot(sensitivity)
\end{lstlisting}
\vspace{1ex}
\textbf{Key question:} How strong would an \emph{unobserved} confounder need to be (relative to the observed confounder $Z$) to explain away the estimated effect of $D$?
\end{frame}

% Slide 4: Interpreting the Diagnostic Outputs
\begin{frame}{Interpreting the Sensitivity Analysis Output}
\textbf{Key Diagnostics:}
\begin{itemize}
    \item \textbf{Robustness Value:} The minimum strength (in terms of partial \(R^2\)) that an unobserved confounder must have with both the treatment and outcome to fully explain away the estimated effect.
    \item \textbf{Benchmark Comparison:} Using the observed \(Z\) as a benchmark helps gauge whether such a confounder is plausible. If an omitted variable would need to be 3$\times$ stronger than \(Z\), the result is likely robust.
    \item \textbf{Contour Plot:} Visualizes all combinations of $R^2_{D\sim U|X}$ and $R^2_{Y\sim U|D,X}$ that would reduce the estimated effect to zero.
\end{itemize}
\end{frame}
