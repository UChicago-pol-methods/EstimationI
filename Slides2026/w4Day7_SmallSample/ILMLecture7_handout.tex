\section{Likelihood}

\setcounter{subsection}{1}

\begin{frame}{Where Are We? \emph{Efficient Estimation}}
\begin{itemize}
\item So far: OLS estimates $\beta$ and we have its variance, but \emph{how precise can any estimator be?}\pause
\item The general theory of \textbf{efficient estimation} answers this via \textbf{scores} (derivatives of log-densities) and \textbf{information} (their variance).\pause
\item These concepts are not specific to MLE --- they underlie influence functions, semiparametric efficiency bounds, and GMM.\pause
\item \textbf{This lecture}: when we know the full likelihood, MLE is the estimator that achieves the efficiency bound.  We use the normal linear model as the leading case.\pause
\item \textbf{The punchline}: OLS \emph{is} the MLE under normality, and its score equations are moment conditions --- the same objects that drive GMM and robust inference.
\end{itemize}
\end{frame}



\begin{frame}{Parametric Model}
\begin{definition}
A \textbf{parametric model} is a family of probability distributions $\{f(x|\theta):\theta\in\Theta\}$ indexed by a finite-dimensional parameter vector $\theta$.
\end{definition}\pause
\begin{itemize}
\item In the continuous case, $f(x|\theta)$ is a probability density function.\pause
\item E.g.\ if $X\sim N(\mu,\sigma^2)$, $f(x|\mu,\sigma^2)=\sigma^{-1}\phi((x-\mu)/\sigma)$, the parameters are $\mu\in \mathbb{R}$ and $\sigma^2>0$.
\item Recall, $\phi(z)$ is the density for the standard normal, $\phi(z)=\frac{1}{\sqrt{2\pi}}\exp(-\frac{z^2}{2})$
\end{itemize}
\end{frame}

\begin{frame}{Specified Parametric Models}
\begin{itemize}
\item A model is called \textbf{correctly specified} when there exists a unique parameter value $\theta_0$ such that $f(x|\theta_0)=f(x)$, the true data distribution.\pause
\item For example, if the true density is $$f(x)=2\exp(-2x)$$
\begin{itemize}
\item the exponential model $f(x|\lambda)=\lambda^{-1}\exp(-x/\lambda)$ is a correctly specified model with $\lambda_0=1/2$\pause
\item the lognormal model $f(x|\mu,\ \sigma^2)=\frac{1}{x\sigma\sqrt{2\pi}}\exp(-\frac{(\ln x-\mu)^2}{2\sigma^2})$ is misspecified, cannot equal $2 \exp(-2x)$ under any parameter value.
\end{itemize}
\end{itemize}

\end{frame}



\begin{frame}{Likelihoods}
\begin{itemize}
\item The density $f(x|\theta)$ tells us: for a fixed $\theta$, how probable is a particular observation $x$?\pause
\item The \textbf{likelihood} asks the reverse question: for fixed observed data $\bm{X}$, how plausible is each $\theta$?
$$\mathcal{L}(\theta)\equiv f(X_1, \ldots, X_n|\theta) \quad\text{(same formula, viewed as a function of }\theta\text{)}$$\pause
\item Example: Binomial model of how many terms ($x$ out of $n=2$) each of 5 candidates wins.  $P(x|p)={n\choose x}p^{x}(1-p)^{n-x}$.\pause
\item Suppose we observe $\bm{x}= \{1,\ 0,\ 1,\ 2,\ 0\}$.\pause
\item The Joint Likelihood of the data is:
\begin{align*}
P(\bm{x}|\ p)&={2\choose 1}p^{1}(1-p)^{1} {2\choose 0}p^{0}(1-p)^{2} {2\choose 1}p^{1}(1-p)^{1}  {2\choose 2}p^{2}(1-p)^{0} {2 \choose 0}p^{0}(1-p)^{2}\\\pause
&=4p^{4}(1-p)^{6}
\end{align*}
\end{itemize}
\end{frame}

\begin{frame}{Maximum Likelihood Estimator (from Hansen Probability)}
\begin{definition}
The \textbf{maximum likelihood estimator} $\hat{\theta}$ of $\theta$ is the value that maximizes the likelihood: $$\mathcal{L}_n(\theta)\equiv f(X_1, X_2, \ldots X_n|\theta)$$
\end{definition}
Call $\mathcal{l}_n=\sum_{i=1}^n \log f(X_i|\theta)$ the log-likelihood.\\
\end{frame}

\begin{frame}{Maximizing the likelihood}
\begin{align*}
P(\bm{x}|\ p)&=4p^{4}(1-p)^{6}\\\pause
\frac{\partial}{\partial p}P(\bm{x}|\ p)&=16p^3(1-p)^6-24p^4(1-p)^5\\\pause
&=0\\\pause
3p^4(1-p)^5&=2p^3(1-p)^6\\\pause
3p&=2(1-p)\\\pause
5p&=2\\\pause
p^*&=\frac{2}{5}
\end{align*}
\end{frame}

\begin{frame}{From MLE to Precision}
We now have an estimator (MLE) and a useful property (invariance).  \textbf{Two questions remain}:\pause
\begin{enumerate}\setlength{\itemsep}{8pt}
\item \textbf{How precise is the MLE?}  We need a measure of information in the data.\pause
\item \textbf{How precise can \emph{any} estimator be?}  We need a lower bound on variance.
\end{enumerate}\pause
\medskip
The answers come from three linked concepts:
\begin{itemize}
\item \textbf{Score}: the derivative of the log-density (the ``gradient'' that MLE sets to zero).
\item \textbf{Fisher Information}: the variance of the score (how much the data tells us).
\item \textbf{Cram\'{e}r-Rao Bound}: no unbiased estimator can beat information$^{-1}$.
\end{itemize}
\end{frame}

\begin{frame}{Score: the slope of the log likelihood}

\begin{itemize}

\item The \textbf{likelihood score} is the derivative of the log-likelihood function:
$$S_n(\theta)=\frac{\partial}{\partial \theta} \mathcal{l}_n(\theta)$$\pause
\item The score is a function of $\theta$ and tells us how sensitive the log-likelihood is to the parameter, and equals zero at the optimum.\pause
\item The \textbf{efficient score} is the derivative of the log likelihood for a single observation, evaluated at $\bm{x}=X_1, X_2,\ldots X_n$ and the true parameter vector
$$S=\frac{\partial}{\partial \theta}\log f(\bm{x}|\theta_0)$$\pause
\item The efficient score fixes $\theta$ at the true value $\theta_0$.
\end{itemize}
\end{frame}



\begin{frame}{Hessian: the curvature of the log likelihood}
\begin{itemize}
\item The \textbf{likelihood Hessian} is the negative second derivative:
$$\mathcal{H}_n(\theta)=-\frac{\partial^2}{\partial \theta \partial \theta'}\mathcal{l}_n(\theta)$$
\item The Hessian matrix is used to calculate the variance.
\end{itemize}
\end{frame}


\begin{frame}{Fisher Information}

\begin{itemize}
\item The \textbf{Fisher information} is the variance of the efficient score (score evaluated at true $\theta_0$).
$$\mathcal{I}_\theta=\mathbb{E}[SS']$$\pause
\item The \textbf{expected Hessian} is the expectation of the Hessian for a single observation:
$$\mathcal{H}_\theta=-\mathbb{E}[\frac{\partial^2}{\partial \theta \partial \theta'}\log f(X|\theta)]$$\pause
\item If the model is correctly specified:
$$\mathcal{I}_\theta=\mathcal{H}_\theta$$
\item \textbf{Intuition}: At $\theta_0$, the score has mean zero. Both $\mathcal{I}$ (score variance) and $\mathcal{H}$ (expected curvature) measure second-order properties of the same density---they agree because these are two ways of measuring the same thing.
\end{itemize}
\end{frame}


\begin{frame}{Cram\'{e}r-Rao Lower Bound}
\begin{itemize}
\item If $\tilde{\theta}$ is \emph{any} unbiased estimator of $\theta$, then:
$$\text{Var}[\tilde{\theta}]\geq (n\mathcal{I}_\theta)^{-1}$$
No unbiased estimator can have variance below the inverse of the total Fisher information.\pause
\item \textbf{Intuition}: More informative data (higher $\mathcal{I}_\theta$) $\to$ tighter bound $\to$ more precise estimation is possible. The bound is an information-theoretic limit on what the data can tell you.\pause
\item An estimator that achieves the bound is called \textbf{efficient}.
\end{itemize}
\end{frame}

\begin{frame}{Cram\'{e}r-Rao vs.\ Gauss-Markov}
\begin{itemize}
\item \textbf{Gauss-Markov} (Lecture 5): OLS is the best \emph{linear} unbiased estimator, under homoskedasticity.  A result specific to the linear model.\pause
\item \textbf{Cram\'{e}r-Rao}: No unbiased estimator---linear or not---can beat the information bound. A general result for \emph{any parametric model}.\pause
\item Under normality, OLS achieves the CR bound $\to$ efficient among \alert{all} unbiased estimators, not just linear ones.\pause
\item \textbf{The broader point}: The CR bound is one instance of a \emph{semiparametric efficiency bound}.  In later lectures, we will see that the same logic---scores, information, influence functions---defines efficient estimators even without a full parametric model.
\end{itemize}
\end{frame}



\begin{frame}{Variance Estimators}

\begin{itemize}
\item The sample Hessian Estimator depends on calculating the second derivatives of the log-likelihood:
$$\hat{\mathcal{H}}_\theta=\frac{1}{n}\sum_{i=1}^n - \frac{\partial}{\partial \theta\partial \theta'} \log f(X_i|\hat{\theta})$$
$$\hat{\mathbf{V}}_1=\hat{\mathcal{H}}_\theta^{-1}$$\pause
 \item The Outer Product Estimator is based on the Fisher Information:
 $$\hat{\mathcal{I}}_\theta=\frac{1}{n}\sum_{i=1}^n(\frac{\partial}{\partial \theta} \log f(X_i|\hat{\theta}))(\frac{\partial}{\partial \theta} \log f(X_i|\hat{\theta}))'$$
$$\hat{\mathbf{V}}_2=\hat{\mathcal{I}}_\theta^{-1}$$

 \end{itemize}
\end{frame}




\begin{frame}{Example: Normal with known mean, unknown variance}
$X\sim N(0, \theta)$, where $\theta\equiv \sigma^2$. $\mathbb{E}[X]=0$, $\mathbb{E}[X^2]=\theta_0$, $\mathbb{E}[X^4]=3\theta_0^2$.\\
The density is:
 $$f(x|\theta)=\frac{1}{(2\pi\theta)^{1/2}}\exp(-\frac{x^2}{2\theta})$$
The log density is:
$$\log f(x|\theta)=-\frac{1}{2}\log(2\pi)-\frac{1}{2}\log(\theta)-\frac{x^2}{2\theta}$$
the first and second derivatives are:
$$\frac{d}{d\theta}\log f(x|\theta)=-\frac{1}{2\theta}+\frac{x^2}{2\theta^2}=\frac{x^2-\theta}{2\theta^2}$$
$$\frac{d^2}{d\theta^2}\log f(x|\theta)=\frac{1}{2\theta^2}-\frac{x^2}{\theta^3}$$

\end{frame}


\begin{frame}{Example: Normal log likelihood}
$$\mathcal{l}_n(\theta)=\sum_{i=1}^n \log f(X_i|\theta)=-\frac{n}{2}\log(2\pi)-\frac{n}{2}\log(\theta)-\frac{1}{2\theta}\sum_{i=1}^n X_i^2$$
$$\frac{d}{d\theta}\mathcal{l}_n(\theta)=-\frac{n}{2\theta}+\frac{1}{2\theta^2}\sum_{i=1}^n X_i^2$$
The MLE is $\hat{\theta}=\frac{1}{n}\sum_{i=1}^n X_i^2$
\end{frame}


\begin{frame}{Example: Score and Fisher Information}
$$\frac{d}{d\theta}\log f(x|\theta)=\frac{x^2-\theta}{2\theta^2}$$
The efficient score $S=\frac{x^2-\theta_0}{2\theta_0^2}$
$$var[S]=\frac{\mathbb{E}[(X^2-\theta_0)^2]}{4\theta_0^4}=\frac{\mathbb{E}[X^4-2X^2\theta_0+\theta_0^2]}{4\theta_0^4}=\frac{3\theta_0^2-2\theta_0^2+\theta^2_0}{4\theta_0^4}=\frac{1}{2\theta_0^2}$$
\end{frame}





\begin{frame}{Example: Information Equality $\mathcal{I}_\theta=\mathcal{H}_\theta$}

$$\mathcal{H}_\theta=\mathbb{E}[-\frac{d^2}{d\theta^2} \log f(X|\theta_0)]=\mathbb{E}[-\frac{d}{d\theta}\frac{x^2-\theta_0}{2\theta_0^2}]=\mathbb{E}[\frac{2x^2-\theta_0}{2\theta_0^3}]=\frac{1}{2\theta_0^2}$$

$$\mathcal{I}_\theta=var[S]=\frac{1}{2\theta_0^2}=\mathcal{H}_\theta$$
The total Fisher information for $n$ observations is $I_n(\theta_0)=n\times \mathcal{I}_\theta=\frac{n}{2\theta^2_0}$\\
The variance of $\hat{\theta}$ is $\frac{1}{I_n(\theta_0)}=2\theta^2_0/n$\\
So we can get the plug-in estimator for the standard error as $\sqrt{2\hat{\theta}^2/n}$.
\end{frame}

\begin{frame}[fragile]{Simulation: Fisher Information Standard Errors}
\begin{lstlisting}[style=Rstyle]

\end{frame}


\begin{frame}{What If the Model Is Wrong? Robust Variance}
Under correct specification, $\mathcal{I}_\theta=\mathcal{H}_\theta$ and either one gives the variance.\\
\textbf{Under misspecification}, they diverge:\pause
\begin{itemize}
\item $\mathcal{H}$ (curvature) still captures how the estimator behaves locally.
\item $\mathcal{I}$ (score variance) captures the actual variability in the data.\pause
\item The correct variance requires \emph{both}:
$$\bm{\hat{V}}=\hat{\mathcal{H}}^{-1}\hat{\mathcal{I}}\,\hat{\mathcal{H}}^{-1}$$\pause
\end{itemize}
\begin{tcolorbox}[colback=blue!5, colframe=blue!50]
This is the \textbf{sandwich formula} --- the same structure as the heteroskedasticity-robust variance from Lecture 6, now derived from likelihood theory.
\end{tcolorbox}
\end{frame}

\begin{frame}{Summary: Scores, Information, and Efficiency}
These four concepts are \alert{not specific to MLE} --- they recur throughout estimation theory:\pause
\begin{enumerate}\setlength{\itemsep}{4pt}
\item \textbf{Score} $\to$ the derivative of the log-density.  Setting it to zero gives estimating equations --- the same objects that define moment conditions and influence functions.\pause
\item \textbf{Fisher Information} $\to$ the variance of the score; measures how much the data can tell you about $\theta$.\pause
\item \textbf{Efficiency bound} $\to$ no unbiased estimator can beat $(n\mathcal{I}_\theta)^{-1}$.  MLE achieves this when the likelihood is correctly specified.\pause
\item \textbf{Likelihood} $\to$ when a full parametric model is available, the likelihood gives us a concrete score and a concrete bound.
\end{enumerate}\pause
\medskip
\textbf{Next}: apply this to the normal linear model and show that OLS \emph{is} the MLE.
\end{frame}

\section{Normal Linear Model}

\begin{frame}{Likelihood methods for linear model}
\begin{itemize}
\item The Normal Regression model assumes that $y\sim N(\mu_Y, \sigma^2)$, or that $e\sim N(0, \sigma^2)$.
\begin{align*}
f(y|\bm{x})&=\frac{1}{(2\pi \sigma^2)^{1/2}}exp\left(-\frac{1}{2\sigma^2}(y-\bm{x}'\bm{\beta})^2\right)\\
f(y_1,\ldots y_n|\bm{x}_1,\ldots \bm{x}_n)&=\prod_{i=1}^n f(y_i|x_i)\\
&=\prod_{i=1}^n\frac{1}{(2\pi \sigma^2)^{1/2}}exp\left(-\frac{1}{2\sigma^2}(y_i-\bm{x}_i'\bm{\beta})^2\right)\\
&=\frac{1}{(2\pi \sigma^2)^{n/2}}\exp\left(-\frac{1}{2\sigma^2}\sum_{i=1}^n(y_i-\bm{x}_i'\bm{\beta})^2\right)\\
&\equiv \mathcal{L}_n(\beta, \sigma^2)
\end{align*}
\end{itemize}
\end{frame}

\begin{frame}{Log Likelihood of the normal linear model}
$$\log \mathcal{L}_n(\beta, \sigma^2) = -\frac{n}{2}log(2\pi \sigma^2)-\frac{1}{2\sigma^2}\sum_{i=1}^n (Y_i-\bm{x}_i'\beta)^2\equiv \mathcal{l}_n(\beta,\sigma^2)$$
\end{frame}



\begin{frame}{Classic Normal Regression Model}
\begin{itemize}
\item The Classic Normal Regression Model consists of the following assumptions:
\begin{enumerate}
\item $\bm{y}=\bm{X\beta}+\bm{e}$
\item $\bm{e}|\bm{X} \sim N(\bm{0},\ \sigma^2\bm{I})$
\begin{itemize}
\item or equivalently: $\bm{y}\sim N(\bm{X\beta},\ \sigma^2\bm{I})$
\end{itemize}
\item rank($\bm{X})=K$.
\end{enumerate}
In this model, $N(\bm{\mu},\ \bm{\Sigma})$ is the multivariate normal density function whose pdf is:
$$f(\bm{e})=(2\pi)^{-n/2}(\sigma^2)^{-n/2}\exp(-\frac{1}{2\sigma^2}\bm{e}'\bm{e})$$
\end{itemize}
\end{frame}

\begin{frame}{Background on Multivariate Normal}
\begin{itemize}
\item A multivariate normal's parameters are a vector of means and a variance covariance matrix.
\item  Linear functions of a multinormal vector $\bm{y}$ are also normal.\\
\item If $\bm{z}=\bm{g}+\bm{H}\bm{y}$, where $\bm{g}$ and $\bm{H}$ are non-random, and $\bm{H}$ has full row rank, then 
$$\bm{z}\sim N(\bm{g}+\bm{H\mu},\ \bm{H\Sigma H'})$$
\item In the case of MVN, independence is the same as uncorrelated.
\item \textbf{See handout}: ``The Bivariate Normal CEF'' derives the conditional distribution $Y\mid X=x$ from the joint density and shows that the CEF is linear with slope $\text{Cov}(Y,X)/\text{Var}(X)$ under joint normality.
\end{itemize}
\end{frame}

\section{Normal MLE}

\begin{frame}{Roadmap: MLE as a Special Case}
\begin{itemize}\setlength{\itemsep}{6pt}
\item We have the normal linear model: $\bm{y}|\bm{X}\sim N(\bm{X\beta},\ \sigma^2\bm{I})$.\pause
\item Solve for the MLE and show it equals OLS --- the full likelihood buys us nothing new for $\hat{\beta}$.\pause
\item The \alert{score} of the normal model \emph{is} the OLS moment condition $\bm{X'e}=0$.  This is the key link: MLE is just-identified GMM with score moments.\pause
\item The payoff of normality: exact finite-sample distributions ($t$, $\chi^2$, $F$) that do not require asymptotics.
\end{itemize}
\end{frame}

\begin{frame}{Solving for the MLE}
\begin{itemize}
\item Given the likelihood, we can solve for the MLE $(\hat{\beta}_{MLE},\ \hat{\sigma}^2_{MLE})$
\begin{align*}
\frac{\partial}{\partial \beta}[-\frac{n}{2}\log(2\pi\sigma^2)-\frac{1}{2\sigma^2}\sum_{i=1}^n(Y_i-\bm{x}_i'\beta)^2]\\\pause
\frac{2}{2\hat{\sigma}_{MLE}^2}\sum_{i=1}^nX_i(Y_i-\bm{x}_i'\hat{\beta}_{MLE})=0\\\pause
\frac{\partial}{\partial \sigma^2}[-\frac{n}{2}\log(2\pi\sigma^2)-\frac{1}{2\sigma^2}\sum_{i=1}^n(Y_i-\bm{x}_i'\beta)^2]\\\pause
-\frac{n}{2}\frac{1}{\hat{\sigma}_{mle}^2}+\frac{1}{2\hat{\sigma}_{mle}^4}\sum_{i=1}^n(Y_i-\bm{x}_i'\hat{\beta}_{MLE})^2=0
\end{align*}
\end{itemize}
\end{frame}

\begin{frame}{Overall Likelihood}
\begin{itemize}
\item $\hat{\beta}_{MLE}=(\sum_{i=1}^n\bm{x}_i\bm{x}'_i)^{-1}(\sum_{i=1}^n\bm{x}_iY_i)=\hat{\beta}_{ols}$.\pause
\item $\hat{\sigma}^2_{MLE}=\frac{1}{n}\sum_{i=1}^n(Y_i-\bm{x}'_i\hat{\beta}_{MLE})^{2}$.  Note: this divides by $n$, not $n{-}k$, so the MLE is \alert{biased} for $\sigma^2$.  The unbiased estimator $s^2$ divides by $n{-}k$.\pause
\item $\log\mathcal{L}(\hat{\beta}_{MLE},\ \hat{\sigma}^2_{MLE})=-\frac{n}{2}\log(2\pi \hat{\sigma}^2_{mle})-n/2$\pause
\item You will see the "log likelihood" reported as a measure of fit, logLik$()$\pause
\item The Akaike's AIC is $-2$log-likehood$+2n_{par}$, weighing model performance versus complexity.\pause
\item AIC gives an asymptotically unbiased estimator of the expected relative Kullback-Leibler divergence when approximating an unknown distribution and a given model.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{R Example: Log-Likelihood and AIC}
\begin{lstlisting}[style=Rstyle]
data(swiss)
mod1 <- lm(Fertility ~ Education, data = swiss)
mod2 <- lm(Fertility ~ Education + Agriculture,
           data = swiss)
mod3 <- lm(Fertility ~ Education + Agriculture
           + Catholic + Infant.Mortality, data=swiss)
# Log-likelihoods (higher = better fit)
sapply(list(mod1, mod2, mod3), logLik)
#  -168.3   -166.0   -155.3
# AIC = -2*logLik + 2*k (lower = better)
sapply(list(mod1, mod2, mod3), AIC)
#   342.5    340.0    322.6
\end{lstlisting}
\smallskip
AIC penalizes complexity: mod3 wins because the likelihood improvement outweighs the $2k$ penalty.
\end{frame}

\begin{frame}{Score of the Normal Regression Model (Hansen 5.14)}
The likelihood scores are the derivatives of the log-likelihood:
\begin{align*}
\frac{\partial}{\partial \beta}\ell_n(\beta,\sigma^2)&=\frac{1}{\sigma^2}\sum_{i=1}^n X_i(Y_i-X_i'\beta)=\frac{1}{\sigma^2}\bm{X'e}\\\pause
\frac{\partial}{\partial \sigma^2}\ell_n(\beta,\sigma^2)&=-\frac{n}{2\sigma^2}+\frac{1}{2\sigma^4}\sum_{i=1}^n(Y_i-X_i'\beta)^2
\end{align*}\pause
Setting the score for $\beta$ to zero:
$$\frac{1}{\sigma^2}\bm{X'e}=0 \quad\Longleftrightarrow\quad \bm{X'e}=0 \quad\Longleftrightarrow\quad \bm{X'(Y-X\beta)}=0$$\pause
\medskip
These are exactly the \alert{OLS normal equations}!  The MLE for $\beta$ equals the OLS estimator because the score FOC is proportional to the least squares FOC.
\end{frame}

\begin{frame}{Scores as Moment Conditions}
The score for $\beta$ evaluated at the true parameter is:
$$S_i(\beta_0)=\frac{\partial}{\partial \beta}\log f(Y_i|X_i,\beta_0,\sigma^2)=\frac{1}{\sigma^2}X_i e_i$$\pause
The population moment condition from the CEF is $\mathbb{E}[X_i e_i]=0$.\pause

\medskip
\textbf{Key observation}: Setting the score to zero in the sample,
$$\frac{1}{n}\sum_{i=1}^n S_i(\hat{\beta})=0 \quad\Longleftrightarrow\quad \frac{1}{n}\sum_{i=1}^n X_i\hat{e}_i=0$$
is the \alert{same} as solving the sample moment condition $\frac{1}{n}\sum X_i(Y_i - X_i'\hat{\beta})=0$.\pause

\smallskip
MLE, OLS, and method of moments all give the \emph{same} estimator for $\beta$ in the normal linear model.
\end{frame}

\begin{frame}{Information Matrix of Normal Regression (Hansen 5.14)}
The Fisher information matrix for the normal regression model is:
$$\mathcal{J}=\text{var}\begin{bmatrix}\frac{\partial}{\partial \beta}\ell(\beta,\sigma^2)\\\frac{\partial}{\partial \sigma^2}\ell(\beta,\sigma^2)\end{bmatrix}=\begin{pmatrix}\frac{1}{\sigma^2}\bm{X'X} & \bm{0}\\\bm{0} & \frac{n}{2\sigma^4}\end{pmatrix}$$\pause
\begin{itemize}
\item The matrix is \alert{block diagonal}: estimation of $\beta$ and $\sigma^2$ are independent.
\item The Cram\'{e}r-Rao lower bound for $\beta$ is:
$$\mathcal{J}_\beta^{-1}=\sigma^2(\bm{X'X})^{-1}$$
which is exactly the variance of $\hat{\beta}_{OLS}$.\pause
\item OLS \alert{achieves the Cram\'{e}r-Rao bound} --- it is efficient among all unbiased estimators under normality.
\end{itemize}
\end{frame}

\begin{frame}{From Scores to the Sandwich (connecting to Lecture 6)}
Under \textbf{correct specification} ($e\sim N(0,\sigma^2)$):
$$\mathcal{I}_\theta=\mathcal{H}_\theta \quad\Longrightarrow\quad V=\mathcal{H}^{-1}=\mathcal{I}^{-1}=\sigma^2(\bm{X'X})^{-1}$$
One formula suffices.  This is the classical OLS variance.\\\pause
\medskip
Under \textbf{misspecification} (heteroskedasticity, non-normality):
$$\mathcal{I}_\theta\neq\mathcal{H}_\theta \quad\Longrightarrow\quad V=\mathcal{H}^{-1}\mathcal{I}\mathcal{H}^{-1}$$\pause
\begin{itemize}
\item The ``bread'' $\mathcal{H}^{-1}$ $\to$ $(\bm{X'X})^{-1}$
\item The ``meat'' $\mathcal{I}$ $\to$ $\bm{X'DX}$ where $D=\text{diag}(\sigma_1^2,\ldots,\sigma_n^2)$
\item The sandwich: $(\bm{X'X})^{-1}(\bm{X'DX})(\bm{X'X})^{-1}$
\end{itemize}\pause
This is the \alert{heteroskedasticity-robust variance} from Lecture 6 (HC0--HC3)!\\
The ``sandwich'' estimator is the likelihood-based variance under misspecification.
\end{frame}

\begin{frame}{MLE as a Special Case of Moment-Based Estimation}
\textbf{Maximum likelihood} solves $k$ score equations in $k$ unknowns:
$\frac{1}{n}\sum_{i=1}^n \frac{\partial}{\partial \theta}\log f(Y_i|X_i,\theta)=0$\pause\\[4pt]
\textbf{GMM} solves $m$ moment conditions in $k$ unknowns:
$\frac{1}{n}\sum_{i=1}^n g(Y_i, X_i, \theta)=0$\pause
\begin{itemize}\setlength{\itemsep}{2pt}
\item When $m=k$ and $g=$ score: \alert{GMM $=$ MLE}.  MLE is just-identified GMM.\pause
\item When $m>k$: GMM uses an \emph{optimal weighting matrix} to combine them.
\item IV example: $\mathbb{E}[Z_i e_i]=0$ gives more moment conditions than parameters.\pause
\end{itemize}
\smallskip
\textbf{Key insight}: MLE $\subset$ GMM.  Score equations are one choice of moment conditions.  When the likelihood is misspecified, the sandwich variance replaces the information-based one.
\end{frame}

\section{Finite-Sample Distributions}

\begin{frame}{The Payoff of Normality: Exact Distributions}
So far we have shown:
\begin{itemize}
\item OLS $=$ MLE under normality, achieving the Cram\'{e}r-Rao bound.
\item The sandwich variance connects likelihood theory to robust inference.
\end{itemize}\pause
\medskip
\textbf{Now}: the normality assumption buys us something no other approach can --- \alert{exact finite-sample distributions} for every quantity of interest:\pause
\begin{enumerate}\setlength{\itemsep}{4pt}
\item $\hat{\bm{\beta}}\sim N$ \hfill (linear function of normal $\bm{e}$)
\item $s^2 \sim \chi^2$ \hfill (quadratic form in normal $\bm{e}$)
\item $T \sim t_{n-k}$ \hfill (ratio of independent normal and $\chi^2$)
\item $F \sim F_{q,n-k}$ \hfill (likelihood ratio test)
\end{enumerate}\pause
\medskip
These hold for \emph{any} $n$ --- no large-sample approximation needed.
\end{frame}

\begin{frame}{Proof of normality of $\bm{\hat{\beta}}$ in CNRM}
We can show that $\bm{\hat{\beta}}\sim N(\bm{\beta},\ \sigma^2\bm{(X'X)}^{-1})$
\begin{align*}
\bm{\hat{\beta}}&=\bm{(X'X)}^{-1}\bm{X'}\bm{y}\\\pause
\bm{\hat{\beta}}&=\bm{(X'X)}^{-1}\bm{X'}(\bm{X}\bm{\beta}+\bm{e})\\\pause
\bm{\hat{\beta}}&=\bm{(X'X)}^{-1}\bm{X'}\bm{X}\bm{\beta}+\bm{(X'X)}^{-1}\bm{X'}\bm{e}\\\pause
\bm{\hat{\beta}}&=\bm{\beta}+\bm{(X'X)}^{-1}\bm{X'}\bm{e}\\\pause
\bm{\hat{\beta}}-\bm{\beta}&=\bm{(X'X)}^{-1}\bm{X'}\bm{e}
\end{align*}

$$\text{Rank}\bm{((X'X)}^{-1}\bm{X'})=\text{Rank}(\bm{X})=K$$
So $\bm{\hat{\beta}}-\bm{\beta}$ is a full-row-rank linear transformation of the multinormal vector $\bm{e}$
$$\bm{\hat{\beta}}-\bm{\beta} \sim N(\bm{0},\ \sigma^2\bm{(X'X)}^{-1}\bm{X'}\bm{X}\bm{(X'X)}^{-1})$$
\end{frame}



\begin{frame}{Distribution of Residuals (Hansen 5.7)}
Recall $\bm{\hat{e}}=\bm{Me}$ where $\bm{M}=\bm{I}_n-\bm{X}(\bm{X'X})^{-1}\bm{X'}$.\\\medskip
Since $\bm{e}|\bm{X}\sim N(\bm{0},\sigma^2\bm{I}_n)$ and $\bm{\hat{e}}$ is a linear function of $\bm{e}$:
$$\bm{\hat{e}}|\bm{X}\sim N(\bm{0},\ \sigma^2\bm{M})$$\pause
\medskip
\textbf{Key fact}: $\hat{\bm{\beta}}$ and $\hat{\bm{e}}$ are \alert{independent} (conditional on $\bm{X}$).\\[4pt]
\emph{Proof}: Their joint covariance is
$$\text{Cov}(\hat{\bm{\beta}}-\bm{\beta},\ \hat{\bm{e}})=(\bm{X'X})^{-1}\bm{X'}\cdot\sigma^2\bm{I}\cdot\bm{M}=\sigma^2(\bm{X'X})^{-1}\underbrace{\bm{X'M}}_{=\bm{0}}=\bm{0}$$
Since they are jointly normal and uncorrelated, they are independent.\\\pause
\medskip
\textbf{Why this matters}: It means $s^2=\hat{\bm{e}}'\hat{\bm{e}}/(n-k)$ is independent of $\hat{\bm{\beta}}$.  This is what allows us to form the $t$-statistic as a ratio of independent normal and $\chi^2$ components.
\end{frame}

\begin{frame}{Distribution of residual sum of squares $s_{\hat{e}}^2=\frac{\bm{\hat{e}'\hat{e}}}{(N-K)}$  in CNRM}
Thm: Let $\bm{z}\sim N(\bm{0},\ \bm{I})$, let $\bm{A}$ be idempotent.  Then $\bm{z}'\bm{A}\bm{z}\sim \chi^2(\nu)$ where $\nu=\text{Rank}(\bm{A})$.
\begin{align*}\pause
\bm{e}|\bm{X} &\sim N(\bm{0},\ \sigma^2\bm{I})\\
\bm{e}/\sigma|\bm{X} &\sim N(\bm{0},\ \bm{I}) \pause
\end{align*}
Recall $\bm{\hat{e}}=\bm{Me}$, so:
\begin{align*}
(N-K)s_{\hat{e}}^2&=\bm{\hat{e}}'\bm{\hat{e}}=\bm{e}'\bm{M}\bm{e}\\
\frac{\bm{e}'}{\sigma}\bm{M}\frac{\bm{e}}{\sigma}&\sim \chi^2(N-K) \tag{$\bm{M}$ is idempotent, rank $N-K$}
\end{align*}
That is, in a linear regression model, $\frac{(N-K)s_{\hat{e}}^2}{\sigma^2}\sim \chi^2_{n-k}$
\end{frame}

\begin{frame}[fragile]{R Example: $\chi^2$ Distribution of $s^2$}
\begin{lstlisting}[style=Rstyle]

  y <- X %*% beta + rnorm(n, sd = sqrt(sigma2))
  ehat <- resid(lm(y ~ X - 1))
  sum(ehat^2) / sigma2   # (n-k)*s^2 / sigma^2
})
\end{lstlisting}\pause

\end{lstlisting}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{From Known $\sigma$ to Estimated $s$: Why Heavier Tails}
If $\sigma$ were known, the $j$-th coefficient is exactly normal:
\[
\frac{\hat{\beta}_j - \beta_j}{\sigma\sqrt{[(\bm{X'X})^{-1}]_{jj}}} \sim N(0, 1)
\]\pause
But $\sigma$ is unknown.  Replacing it with $s$ introduces extra randomness:\pause
\begin{itemize}
\item When $s$ underestimates $\sigma$, the ratio is \emph{inflated} --- producing occasional large values.
\item When $s$ overestimates, the ratio is compressed.\pause
\item Net effect: \alert{heavier tails} than the normal, especially when $n{-}k$ is small (because $s$ is noisy with few degrees of freedom).\pause
\item As $n{-}k\to\infty$, $s\to\sigma$ and $t_{n-k}\to N(0,1)$.
\end{itemize}
\end{frame}

\begin{frame}{From Normal to $t$: Estimated Variance}
In practice, we do not know the true error variance \( \sigma^2 \).  
The unbiased estimator is:
\[
s_{\hat{e}}^2 = \frac{1}{n - k} \sum_{i=1}^n \hat{e}_i^2 = \frac{\hat{\bm{e}}' \hat{\bm{e}}}{n - k}
\]

We plug in this estimate to standardize:
\[
T = \frac{\hat{\beta}_j - \beta_j}{\sqrt{s_{\hat{e}}^2 \left[(\bm{X'X})^{-1}\right]_{jj}}}
\]

This is no longer standard normal, but it has a \textbf{$t$-distribution}.
\end{frame}

\begin{frame}{The $t$ Statistic and its Distribution}
Recall: if $Z \sim N(0, 1)$, $W \sim \chi^2(n - k)$, and $Z \perp W$, then $\frac{Z}{\sqrt{W/(n - k)}} \sim t(n - k)$.\pause

\medskip
In our case, $Z=\frac{\hat{\beta}_j - \beta_j}{\sigma\sqrt{[(\bm{X'X})^{-1}]_{jj}}} \sim N(0, 1)$ \quad and \quad $W=\frac{(n - k) s_{\hat{e}}^2}{\sigma^2} \sim \chi^2(n - k)$.\pause

\medskip
Taking the ratio and simplifying ($\sigma^2$ cancels):
\[
T = \frac{\hat{\beta}_j - \beta_j}{\sqrt{s_{\hat{e}}^2 \left[(\bm{X'X})^{-1}\right]_{jj}}}
\sim t(n - k)
\]
\end{frame}

\begin{frame}{Limits of the $t$-Statistic}
\begin{itemize}
\item The $t$-statistic $T=\frac{\hat{\beta}-\beta_0}{s(\hat{\beta})}$ follows $t_{n-k}$ \textbf{exactly} under two conditions: homoskedasticity and i.i.d.\ normality of $e$.\pause
\item Without normality, OLS is still unbiased, but the exact $t_{n-k}$ distribution no longer holds.\pause
\item With HC0--HC3 robust standard errors, we do not have an exact finite-sample distribution for $T$.\pause
\item \textbf{What do we do instead?} We rely on \emph{large-sample approximations}: the CLT guarantees that $T\overset{d}{\to} N(0,1)$ as $n\to\infty$, regardless of the error distribution. This is the subject of Lectures 10--11 (Asymptotics).\pause
\item For now: the normal model gives exact, clean results. Relaxing normality requires asymptotic tools.
\end{itemize}
\end{frame}

\begin{frame}{The $F$-test as a Likelihood Ratio Test (Hansen 5.13)}
Consider testing $\mathbb{H}_0: \beta_2=0$ in $Y=X_1'\beta_1+X_2'\beta_2+e$.\\
The likelihood ratio statistic compares the maximized log-likelihoods:
$$\text{LR}=2\left(\ell_n(\hat{\beta},\hat{\sigma}^2)-\ell_n(\tilde{\beta}_1,\tilde{\sigma}^2)\right)=n\log\!\left(\frac{\tilde{\sigma}^2}{\hat{\sigma}^2}\right)$$\pause
where $\tilde{\sigma}^2$ is from the restricted (null) model.  This is equivalent to the $F$-statistic:
$$F=\frac{(\tilde{\sigma}^2-\hat{\sigma}^2)/q}{\hat{\sigma}^2/(n-k)}\sim F_{q,\, n-k}\quad\text{under }\mathbb{H}_0$$\pause
\begin{itemize}
\item $q=\dim(\beta_2)$ is the number of restrictions.
\item Under the null, $F$ has an \emph{exact} $F$-distribution in the normal model.
\item This justifies the $F$-test reported by \texttt{anova()} and regression output.\pause
\item The $t$-test is a special case: when $q=1$, $F=T^2$ and $F_{1,n-k}=t_{n-k}^2$.
\end{itemize}
\end{frame}

\begin{frame}{Summary: Small-Sample Distributions}
Under the classical normal regression model:\pause
\begin{itemize}\setlength{\itemsep}{4pt}
\item $\hat{\bm{\beta}}|\bm{X}\sim N(\bm{\beta},\ \sigma^2(\bm{X'X})^{-1})$ \hfill \emph{(linear fn of normal)}\pause
\item $\hat{\bm{e}}|\bm{X}\sim N(\bm{0},\ \sigma^2\bm{M})$, \quad $\hat{\bm{\beta}}\perp\hat{\bm{e}}$ \hfill \emph{(projection)}\pause
\item $\frac{(n-k)s^2}{\sigma^2}\sim \chi^2_{n-k}$ \hfill \emph{(idempotent quadratic form)}\pause
\item $T=\frac{\hat{\beta}_j-\beta_j}{s(\hat{\beta}_j)}\sim t_{n-k}$ \hfill \emph{(normal / $\sqrt{\chi^2/\text{df}}$)}\pause
\item $F=\frac{(\tilde{\sigma}^2-\hat{\sigma}^2)/q}{\hat{\sigma}^2/(n-k)}\sim F_{q,n-k}$ \hfill \emph{(likelihood ratio)}\pause
\end{itemize}
\medskip
These \alert{exact} results hold in finite samples.  Without normality, we rely on asymptotic approximations.
\end{frame}

\section{Hypothesis Testing}

\begin{frame}{Confidence Intervals}
\begin{itemize}
\item $\hat{\beta}$ is a \textbf{point estimate} for a coefficient $\beta$.  \pause
\item We can instead estimate an interval, $\hat{C}=[\hat{L},\ \hat{U}]$ which contains the true value with high probability. \pause
\item An interval estimate $\hat{C}$ is called a $1-\alpha$ confidence interval when $Pr(\beta \in \hat{C})=1-\alpha$.  The value $1-\alpha$ is called the \textbf{coverage probability}. \pause
\item The key mistake is in thinking that the above statement treats $\beta$ as random and $\hat{C}$ is fixed, (the probability that $\beta$ is in $\hat{C}$).   \pause
\item $Pr(\beta \in \hat{C})$ is the probability that the random set $\hat{C}$ covers or contains $\beta$.  \pause
$$\hat{C}=[\hat{\beta}-c\cdot s(\hat{\beta}),\ \hat{\beta}+c\cdot s(\hat{\beta})]$$
\end{itemize}
\end{frame}

\begin{frame}{Confidence Intervals in Practice}
\begin{align*}
Pr(\beta\in \hat{C})&=Pr(-c\leq T(\beta)\leq c)\\
Pr(\beta\in \hat{C})&=2F(c)-1
\end{align*}
Our goal is to set this coverage probability equal to $1-\alpha$, or $F(c)=1-\alpha/2$.\\
If $\alpha=.05$, we solve $c=F^{-1}(1-.05/2)$.  In case of a normal, c=$1.96\approx 2$
$$\hat{C}=[\hat{\beta}-2\cdot s(\hat{\beta}),\ \hat{\beta}+2\cdot s(\hat{\beta})]$$
\end{frame}

\begin{frame}{t test and p-values}
\begin{itemize}
\item  A theory is said to have \emph{testable implications} if it can be falsified. \pause
\item For example, a theory may be false if $\beta=\beta_0$.  This is called a "null hypothesis" $\mathbb{H}_0$. \pause
\item We further specify the complement of $\mathbb{H}_0$ as $\mathbb{H}_1$. \pause
\item A statistic can be informative, some realizations may be unlikely if $\mathbb{H}_0$ is true. \pause
\item Define a test statistic: $|T|=\left| \frac{\hat{\beta}-\beta_0}{s(\hat{\beta})}\right|$ and set a critical value $c$.
$$\text{Reject }\mathbb{H}_0\text{ if } |T|>c$$ 
\item A p-value indexes a test's strength of rejection of the null. \pause
\item In a normal regression model, $p=2(1-F_{n-k}(|T|))$ 
\end{itemize}
\end{frame}

\begin{frame}{Beyond Single Coefficients: Joint Restrictions}
\begin{itemize}
\item The $t$-test asks: is \emph{one} coefficient zero?\pause
\item Many research questions involve \emph{multiple} coefficients simultaneously.\pause
\item \textbf{Example}: Does a job training program affect wages?
$$\text{Wage}_i = \beta_0 + \beta_1 \text{Training}_i + \beta_2 (\text{Training}_i \times \text{Experience}_i) + \beta_3 \text{Experience}_i + e_i$$\pause
\item The effect of training is $\beta_1 + \beta_2 \cdot \text{Experience}_i$ --- it depends on \emph{both} $\beta_1$ and $\beta_2$.\pause
\item To test whether training has \emph{any} effect, we need:
$$\mathbb{H}_0: \beta_1 = 0 \text{ and } \beta_2 = 0$$\pause
\item A $t$-test on $\beta_1$ alone misses the interaction; a $t$-test on $\beta_2$ alone misses the main effect.  We need a \alert{joint test}.
\end{itemize}
\end{frame}

\begin{frame}{The $F$-Test for Joint Restrictions}
To test $\mathbb{H}_0: \bm{R\beta}=\bm{r}$ where $\bm{R}$ is $q\times k$:\pause
\begin{enumerate}\setlength{\itemsep}{6pt}
\item Estimate the \textbf{unrestricted} model: get $\hat{\bm{e}}'\hat{\bm{e}}$ (the full SSR).\pause
\item Estimate the \textbf{restricted} model (imposing $\mathbb{H}_0$): get $\tilde{\bm{e}}'\tilde{\bm{e}}$.\pause
\item The $F$-statistic measures how much fit deteriorates under the restriction:
$$F = \frac{(\tilde{\bm{e}}'\tilde{\bm{e}} - \hat{\bm{e}}'\hat{\bm{e}})/q}{\hat{\bm{e}}'\hat{\bm{e}}/(n - k)} \sim F_{q,\, n-k} \quad\text{under }\mathbb{H}_0$$
\end{enumerate}\pause
\medskip
\textbf{Intuition}: If $\mathbb{H}_0$ is true, dropping the restricted variables should not increase SSR much.  A large $F$ means the restricted model fits \alert{substantially worse} $\to$ reject.\pause\\[4pt]
When $q=1$, this reduces to $F=T^2$ --- the $t$-test is a special case.
\end{frame}

\begin{frame}[fragile]{Example: Joint Test for Treatment Effect}
\begin{lstlisting}[style=Rstyle]
set.seed(42); n <- 80
experience <- runif(n, 1, 20)
training <- rbinom(n, 1, 0.5)
wage <- 10 + 3*training + 0.5*training*experience +
        0.8*experience + rnorm(n, sd = 3)

full <- lm(wage ~ training*experience)       # unrestricted
restricted <- lm(wage ~ experience)           # restricted

anova(restricted, full)  # joint F-test
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Example: Why the Joint Test Matters}
\begin{lstlisting}[style=Rstyle]
summary(full)$coefficients[c("training",
                  "training:experience"), ]
#                    Estimate Std.Err t    Pr(>|t|)
# training              1.81   1.48  1.22  0.225
# training:experience   0.60   0.13  4.64  <0.001

anova(restricted, full)
# F = 39.2, df1 = 2, df2 = 76, p < 0.001
\end{lstlisting}\pause
\smallskip
\begin{itemize}\setlength{\itemsep}{2pt}
\item The $t$-test on \texttt{training} alone: $p=0.225$ --- \alert{not significant}!\pause
\item But the joint $F$-test: $p<0.001$ --- training clearly matters.\pause
\item The interaction absorbs much of the treatment's impact at higher experience levels, so the main effect alone is small.\pause
\item \textbf{Lesson}: When a variable enters with an interaction, always test jointly.
\end{itemize}
\end{frame}

\begin{frame}{Writing Joint Restrictions as $\bm{R\beta}=\bm{r}$}
For the model $\text{Wage}=\beta_0+\beta_1 \text{Tr}+\beta_2 \text{Tr}\!\times\!\text{Exp}+\beta_3\text{Exp}+e$:\pause
$$\mathbb{H}_0:\beta_1=0\text{ and }\beta_2=0 \quad\Longleftrightarrow\quad \underbrace{\begin{pmatrix}0 & 1 & 0 & 0\\0 & 0 & 1 & 0\end{pmatrix}}_{\bm{R}}\begin{pmatrix}\beta_0\\\beta_1\\\beta_2\\\beta_3\end{pmatrix}=\underbrace{\begin{pmatrix}0\\0\end{pmatrix}}_{\bm{r}}$$\pause
The $F$-statistic can also be written directly from the unrestricted estimates:
$$F=\frac{(\bm{R}\hat{\bm{\beta}}-\bm{r})'[\bm{R}\,\hat{V}\,\bm{R}']^{-1}(\bm{R}\hat{\bm{\beta}}-\bm{r})}{q}$$\pause
where $\hat{V}=s^2(\bm{X'X})^{-1}$. This form requires only the \emph{unrestricted} model --- no need to re-estimate under $\mathbb{H}_0$.\\[4pt]\pause
\textbf{General}: any set of $q$ linear restrictions on $\bm{\beta}$ can be tested this way.
\end{frame}
