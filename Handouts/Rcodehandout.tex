\section*{Probability and Distributions}

\begin{tabular}{@{}lll@{}}
\toprule
\textbf{R Code} & \textbf{Math} & \textbf{Description} \\
\midrule
\texttt{dnorm(x, mu, sd)} & $\phi(x;\mu,\sigma)$ & Normal PDF \\
\texttt{pnorm(x, mu, sd)} & $\Phi(x;\mu,\sigma)$ & Normal CDF \\
\texttt{qnorm(p, mu, sd)} & $\Phi^{-1}(p)$ & Normal quantile function \\
\texttt{rnorm(n, mu, sd)} & $X \sim N(\mu,\sigma^2)$ & Draw $n$ normal random variables \\
\addlinespace
\texttt{dt(x, df)}, \texttt{pt()}, \texttt{qt()} & & Student $t$ density, CDF, quantile \\
\texttt{dchisq(x, df)}, \texttt{pchisq()}, \texttt{qchisq()} & & $\chi^2$ density, CDF, quantile \\
\texttt{pf(x, df1, df2)}, \texttt{qf()} & & $F$ distribution CDF, quantile \\
\texttt{dbinom(x, n, p)}, \texttt{rbinom()} & & Binomial PMF, random draws \\
\texttt{plogis(x)}, \texttt{qlogis()} & & Logistic CDF, quantile \\
\addlinespace
\texttt{integrate(f, lower, upper)} & $\int_a^b f(x)\,dx$ & Numerical integration \\
\bottomrule
\end{tabular}

\section*{Descriptive Statistics}

\begin{tabular}{@{}lll@{}}
\toprule
\textbf{R Code} & \textbf{Math} & \textbf{Description} \\
\midrule
\texttt{mean(x)} & $\bar{x} = \frac{1}{n}\sum_i x_i$ & Sample mean \\
\texttt{var(x)} & $\frac{1}{n-1}\sum_i (x_i - \bar{x})^2$ & Sample variance (Bessel-corrected) \\
\texttt{sd(x)} & $\sqrt{\widehat{\text{Var}}(x)}$ & Sample standard deviation \\
\texttt{cov(x, y)} & $\frac{1}{n-1}\sum_i(x_i-\bar{x})(y_i-\bar{y})$ & Sample covariance \\
\texttt{cor(x, y)} & $\widehat{\text{Cor}}(x,y)$ & Sample correlation \\
\texttt{quantile(x, p)} & & Sample quantiles \\
\texttt{length(x)} & $n$ & Number of elements \\
\bottomrule
\end{tabular}

\section*{Matrix Operations}

\begin{tabular}{@{}lll@{}}
\toprule
\textbf{R Code} & \textbf{Math} & \textbf{Description} \\
\midrule

\texttt{diag(A)} & $\text{diag}(\bm{A})$ & Extract diagonal \\
\texttt{diag(n)} & $\bm{I}_n$ & $n\times n$ identity matrix \\
\texttt{det(A)} & $|\bm{A}|$ & Determinant \\
\texttt{dim(A)} & $n \times m$ & Matrix dimensions \\
\texttt{qr(A)\$rank} & $\text{rank}(\bm{A})$ & Matrix rank \\
\texttt{eigen(A)} & & Eigenvalues and eigenvectors \\
\texttt{cbind(x, y)}, \texttt{rbind()} & & Bind columns or rows \\
\bottomrule
\end{tabular}

\section*{OLS Regression}

\begin{tabular}{@{}lll@{}}
\toprule
\textbf{R Code} & \textbf{Math} & \textbf{Description} \\
\midrule
\texttt{lm(y \~{} x1 + x2, data)} & & Fit OLS regression \\
\texttt{coef(mod)} & $\hat{\bm{\beta}} = (\bm{X}'\bm{X})^{-1}\bm{X}'\bm{y}$ & Coefficient estimates \\
\texttt{fitted(mod)} & $\hat{\bm{y}} = \bm{X}\hat{\bm{\beta}}$ & Fitted values \\
\texttt{resid(mod)} & $\bm{e} = \bm{y} - \hat{\bm{y}}$ & Residuals \\
\texttt{vcov(mod)} & $\hat{\sigma}^2(\bm{X}'\bm{X})^{-1}$ & Variance-covariance of $\hat{\bm{\beta}}$ \\
\texttt{sigma(mod)} & $\hat{\sigma} = \sqrt{\bm{e}'\bm{e}/(n-k)}$ & Estimated error std.\ deviation \\
\texttt{confint(mod)} & & Confidence intervals for $\hat{\bm{\beta}}$ \\
\texttt{predict(mod, newdata, interval)} & & Predictions with optional intervals \\
\texttt{summary(mod)} & & Full regression summary \\
\texttt{model.matrix(mod)} & $\bm{X}$ & Extract the design matrix \\
\texttt{update(mod, . \~{} . + x3)} & & Add/remove terms from a model \\
\bottomrule
\end{tabular}
